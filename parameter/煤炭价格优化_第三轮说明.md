## ⚙️ 实现细节

### 多头注意力实现

```python
def create_multi_head_attention(input_tensor, attention_dim, num_heads):
    """多头注意力机制"""
    assert attention_dim % num_heads == 0, "attention_dim必须能被num_heads整除"
    
    head_dim = attention_dim // num_heads
    head_outputs = []
    
    for i in range(num_heads):
        # 每个头独立的Q,K,V
        query = Dense(head_dim, name=f'head_{i}_query')(input_tensor)
        key = Dense(head_dim, name=f'head_{i}_key')(input_tensor)
        value = Dense(head_dim, name=f'head_{i}_value')(input_tensor)
        
        # Scaled Dot-Product Attention
        scores = Dot(axes=[2,2])([query, key])
        scores = Lambda(lambda x: x / tf.math.sqrt(float(head_dim)))(scores)
        attention_weights = Softmax(axis=-1)(scores)
        
        # Apply attention
        context = Dot(axes=[2,1])([attention_weights, value])
        head_outputs.append(context)
    
    # 拼接所有头
    if num_heads > 1:
        multi_head = Concatenate(axis=-1)(head_outputs)
    else:
        multi_head = head_outputs[0]
    
    # 输出投影
    output = Dense(attention_dim, name='multi_head_output')(multi_head)
    output = GlobalAveragePooling1D()(output)
    
    return output
```
